% BibLaTeX loads each entry once per sorting context. Track entry keys so the
% contributed-presentation context cannot inflate the publication total.
\AtDataInput{%
  \ifkeyword{published}
    {\ifcsundef{cvpublishedseen@\thefield{entrykey}}
       {\csgdef{cvpublishedseen@\thefield{entrykey}}{}%
        \stepcounter{cvpublishedtotal}}
       {}}
    {}}
\newcounter{cvpubnumber}
\defbibenvironment{cvreverse}
  {\setcounter{cvpubnumber}{\value{cvpublishedtotal}}%
   \list{}
     {\setlength{\leftmargin}{2.5em}%
      \setlength{\labelwidth}{2em}%
      \setlength{\labelsep}{0.5em}%
      \setlength{\itemsep}{\bibitemsep}%
      \setlength{\parsep}{\bibparsep}}}
  {\endlist}
  {\item[\arabic{cvpubnumber}.]\addtocounter{cvpubnumber}{-1}}

  {\item[\arabic{cvbibnumber}.]\stepcounter{cvbibnumber}}
